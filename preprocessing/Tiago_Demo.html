#Cropped
face_cropper = cropping_face()
skipped = 0

for i in range(20):
    image = images[i]
    label = labels[i]

    w, h, c = image.shape
    
    try:
        
        transformed_images = face_cropper.process(image)

        plt.tight_layout()
        ax = plt.subplot(5, 4, i + 1)
        ax.imshow(transformed_images)
        
        ax.axis("off")
    except SkipImage:
        skipped += 1
        continue



plt.show()
print("Skipped images:", skipped)
